activate EvServ

activate DB

deactivate DB

  deactivate EvServ
  AuthCtrl --> Guest: redirect /verify-email?error=invalid_token
  deactivate AuthCtrl

  EvServ -> EvServ: isUsed() == true
  EvServ --> AuthCtrl: throw IllegalArgumentException("Token already used")
  deactivate EvServ
  AuthCtrl --> Guest: redirect /verify-email?error=token_used
  deactivate AuthCtrl

  EvServ -> EvServ: isExpired() == true
  EvServ --> AuthCtrl: throw IllegalArgumentException("Token expired")
  deactivate EvServ
  AuthCtrl --> Guest: redirect /verify-email?error=token_expired
  deactivate AuthCtrl
else token poprawny
  EvServ -> DB: findUserById(userId)
  activate DB
  DB --> EvServ: user
  deactivate DB
  EvServ -> DB: updateUser(user.setVerified(true).setVerifiedAt(now))
  activate DB
  DB --> EvServ: updated
  deactivate DB
  EvServ -> DB: save(token.setUsed(true))
  activate DB
  DB --> EvServ: saved
  deactivate DB
  EvServ --> AuthCtrl: success
  deactivate EvServ
  AuthCtrl --> Guest: redirect /verify-email?success=true
  deactivate AuthCtrl
end

            startDate, endDate,
            totalPrice, status="PENDING",
            termsId=null
          }
        end note
        deactivate ResCtrl
      end
    end
  end

@startuml
actor "Użytkownik" as User
participant "ToolController" as ToolCtrl
participant "ToolService" as ToolServ
participant "GeoLocationUtil" as GeoUtil
database "Database" as DB

User -> ToolCtrl: GET /api/v1/tools/search
activate ToolCtrl
note right of User
  Query params:
  - search (String, optional)
  - category (String, optional)
  - latitude (Double, optional)
  - longitude (Double, optional)
  - radius (Double, optional, km)
  - page (int, default=0)
  - size (int, default=10)
  - sortBy (String, default="id")
  - sortDirection (String, default="desc")
end note
ToolCtrl -> ToolCtrl: walidacja parametrów
alt niepoprawna kategoria
  ToolCtrl --> User: 400 Bad Request "Invalid category"
  deactivate ToolCtrl
else wyszukiwanie z geolokalizacją
  note right: latitude && longitude != null
  ToolCtrl -> ToolServ: findNearbyTools(lat, lng, radius, search, category, pageable)
  activate ToolServ
  alt search && category
    ToolServ -> DB: findApprovedToolsByNameOrDescriptionAndCategory(search, search, category, pageable)
    activate DB
  else tylko search
    ToolServ -> DB: findApprovedToolsByNameOrDescription(search, search, pageable)
    activate DB
  else tylko category
    ToolServ -> DB: findAllApprovedAndActiveToolsByCategory(category, pageable)
    activate DB
  else brak filtrów
    ToolServ -> DB: findAllApprovedAndActiveTools(pageable)
    activate DB
  end
  note right
    WHERE moderationStatus=APPROVED
    AND isActive=true
  end note
  DB --> ToolServ: List<Tool>
  deactivate DB
  ToolServ -> ToolServ: dla każdego tool
  loop dla każdego tool
    ToolServ -> GeoUtil: calculateDistance(userLat, userLng, toolLat, toolLng)
    activate GeoUtil
    note right: Haversine formula
    GeoUtil --> ToolServ: distance (km)
    deactivate GeoUtil
    alt radius > 0 && distance > radius
      ToolServ -> ToolServ: odrzuć tool
    else distance <= radius || radius <= 0
      ToolServ -> ToolServ: dodaj tool z distance
    end
  end
  ToolServ -> ToolServ: sortowanie według sortBy i sortDirection
  note right
    Możliwe sortBy:
    - distance (default z geo)
    - id, name, pricePerDay
    - createdAt, updatedAt
  end note
  ToolServ -> ToolServ: paginacja (PageImpl)
  ToolServ --> ToolCtrl: Page<Tool>
  deactivate ToolServ
  ToolCtrl -> ToolCtrl: mapowanie do ToolDto + distance
  ToolCtrl --> User: 200 OK (Page<ToolDto> with distance)
  note right
    {
      tools: [...],
      currentPage, totalPages,
      totalItems, pageSize
    }
  end note
  deactivate ToolCtrl
else wyszukiwanie bez geolokalizacji
  alt search && category
    ToolCtrl -> ToolServ: searchActiveTools(search, category, pageable)
    activate ToolServ
    ToolServ -> DB: findApprovedToolsByNameOrDescriptionAndCategory(search, search, category, pageable)
    activate DB
  else tylko search
    ToolCtrl -> ToolServ: searchActiveTools(search, pageable)
    activate ToolServ
    ToolServ -> DB: findApprovedToolsByNameOrDescription(search, search, pageable)
    activate DB
  else tylko category
    ToolCtrl -> ToolServ: getActiveToolsByCategory(category, pageable)
    activate ToolServ
    ToolServ -> DB: findAllApprovedAndActiveToolsByCategory(category, pageable)
    activate DB
  else brak filtrów
    ToolCtrl -> ToolServ: getActiveTools(pageable)
    activate ToolServ
    ToolServ -> DB: findAllApprovedAndActiveTools(pageable)
    activate DB
  end
  note right
    WHERE moderationStatus=APPROVED
    AND isActive=true
    LIKE %search% in name/description
  end note
  DB --> ToolServ: Page<Tool>
  deactivate DB
  ToolServ --> ToolCtrl: Page<Tool>
  deactivate ToolServ
  ToolCtrl -> ToolCtrl: mapowanie do ToolDto (bez distance)
  ToolCtrl --> User: 200 OK (Page<ToolDto>)
  deactivate ToolCtrl
end
@enduml


Diagram stanów 1 – Cykl życia narzędzia

@startuml
state "PENDING" as Pending
state "APPROVED" as Approved
state "REJECTED" as Rejected

[*] --> Pending : Użytkownik dodaje narzędzie\n(moderationStatus=PENDING, isActive=true)

Pending --> Approved : Moderator zatwierdza\napproveTool(toolId, moderatorId, comment)
Pending --> Rejected : Moderator odrzuca\nrejectTool(toolId, moderatorId, comment)
Pending --> Pending : Właściciel edytuje\nupdateTool(toolId, toolUpdateRequest)

Rejected --> Approved : Moderator zatwierdza\napproveTool(toolId, moderatorId, comment)
Rejected --> Pending : Właściciel edytuje\nupdateTool(toolId, toolUpdateRequest)

Approved --> Rejected : Moderator odrzuca\nrejectTool(toolId, moderatorId, comment)
Approved --> Pending : Właściciel edytuje\nupdateTool(toolId, toolUpdateRequest)
Approved --> Approved : Właściciel zmienia isActive\nsetToolStatus(toolId, true/false)

note right of Pending
  moderationStatus = PENDING
  isActive = true (nowe) lub false (edycja)
  moderatorId = null
  moderatedAt = null

  Akcje moderatora:
  - Zatwierdź → APPROVED
  - Odrzuć → REJECTED

  Akcje właściciela:
  - Edytuj → PENDING (ponowna moderacja)

  Niewidoczne (filtr: APPROVED && isActive)
end note

note right of Approved
  moderationStatus = APPROVED
  isActive = true (auto-set)
  moderatorId = <id moderatora>
  moderatedAt = <timestamp>

  Akcje moderatora:
  - Odrzuć → REJECTED

  Akcje właściciela:
  - Zmiana isActive (true ↔ false)
  - Edytuj → PENDING (ponowna moderacja)

  Widoczne gdy isActive=true
end note

note right of Rejected
  moderationStatus = REJECTED
  isActive = false (auto-set)
  moderationComment = <powód odrzucenia>

  Akcje moderatora:
  - Zatwierdź → APPROVED

  Akcje właściciela:
  - Edytuj → PENDING (naprawa + ponowna moderacja)

  Niewidoczne dla użytkowników
end note
@enduml

  - Właściciel → CONFIRMED
  - Najemca lub właściciel → CANCELED

  status = 'CONFIRMED'

  przez najemcę

  - Najemca akceptuje → REGULATIONS_ACCEPTED
  - Najemca lub właściciel → CANCELED
end note
